from __future__ import annotations

from typing import Dict, Optional


class BrokerAdapter:
    def __init__(self, broker_name: str, api_key: Optional[str] = None, api_secret: Optional[str] = None, symbol: Optional[str] = None):
        self.broker_name = broker_name
        self.api_key = api_key
        self.api_secret = api_secret
        self.symbol = symbol

    def get_status(self) -> Dict[str, object]:
        if self.api_key and self.api_secret:
            status = "Ready for market access"
            connected = True
        else:
            status = "Credentials required"
            connected = False

        return {
            "broker": self.broker_name,
            "connected": connected,
            "symbol": self.symbol,
            "status": status,
            "warning": "Only use official exchange-approved APIs and always require manual permission before live orders.",
        }


class BrokerManager:
    @staticmethod
    def get_adapter(broker_name: str, api_key: Optional[str] = None, api_secret: Optional[str] = None, symbol: Optional[str] = None) -> BrokerAdapter:
        return BrokerAdapter(broker_name=broker_name, api_key=api_key, api_secret=api_secret, symbol=symbol)
